    async function youtubeFetch(url,opts={}){if(!hasYouTubeToken())throw new Error('Conecte una cuenta de YouTube antes de continuar.');const response=await fetch(url,{...opts,headers:{Authorization:`Bearer ${state.youtubeToken}`,...(opts.headers||{})}});if(!response.ok){const msg=await response.text();throw new Error(`YouTube respondió ${response.status}: ${msg.slice(0,160)}`);}return response;}

    async function uploadWithRetry(sessionUrl,file){const chunkSize=256*1024;let start=0;while(start<file.size){const end=Math.min(start+chunkSize,file.size);const blob=file.slice(start,end);let done=false;for(let attempt=0;attempt<5&&!done;attempt++){try{const r=await youtubeFetch(sessionUrl,{method:'PUT',headers:{'Content-Type':file.type||'video/*','Content-Range':`bytes ${start}-${end-1}/${file.size}`},body:blob});if(r.status===200||r.status===201){$('uploadProgress').style.width='100%';return await r.json();}if(r.status===308){start=end;done=true;$('uploadProgress').style.width=`${Math.round(start/file.size*100)}%`;$('uploadProgressText').textContent=`Subiendo… ${Math.round(start/file.size*100)} %`;}}catch(error){if(attempt===4)throw error;await new Promise(resolve=>setTimeout(resolve,Math.pow(2,attempt)*800));}}}throw new Error('La sesión de carga terminó sin respuesta de video.');}
    async function uploadYouTube(){
      const file=$('videoFileInput').files[0];
      if(!file)return toast('Falta el video','Seleccione un archivo de video antes de cargar.','warn');
      if(!hasYouTubeToken())return toast('Autorización necesaria','Conecte una cuenta de YouTube antes de cargar.','warn');
      const privacy=$('privacyStatus').value;
      const accepted=window.confirm(`Confirmación requerida: se cargará “${file.name}” a YouTube con visibilidad “${privacy}”. ¿Desea continuar?`);
      if(!accepted)return;
      try{
        $('youtubeUpload').disabled=true;
        $('uploadProgress').style.width='3%';
        $('uploadProgressText').textContent='Creando sesión de carga reanudable…';
        const metadata={snippet:{title:$('uploadTitle').value,description:$('uploadDescription').value,categoryId:$('uploadCategory').value},status:{privacyStatus:privacy}};
        const session=await youtubeFetch('https://www.googleapis.com/upload/youtube/v3/videos?uploadType=resumable&part=snippet,status',{method:'POST',headers:{'Content-Type':'application/json; charset=UTF-8','X-Upload-Content-Type':file.type||'video/*','X-Upload-Content-Length':String(file.size)},body:JSON.stringify(metadata)});
        const sessionUrl=session.headers.get('Location');
        if(!sessionUrl)throw new Error('YouTube no entregó una URL de sesión reanudable.');
        const video=await uploadWithRetry(sessionUrl,file);
        const playlist=$('playlistSelect').value;
        if(playlist){
          await youtubeFetch('https://www.googleapis.com/youtube/v3/playlistItems?part=snippet',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({snippet:{playlistId:playlist,resourceId:{kind:'youtube#video',videoId:video.id}}})});
        }
        if(state.thumbnailBlob){
          await youtubeFetch(`https://www.googleapis.com/upload/youtube/v3/thumbnails/set?videoId=${encodeURIComponent(video.id)}&uploadType=media`,{method:'POST',headers:{'Content-Type':'image/png'},body:state.thumbnailBlob});
        }
        $('uploadProgressText').innerHTML=`Carga completa: <a href="https://www.youtube.com/watch?v=${video.id}" target="_blank" rel="noopener noreferrer">abrir video publicado ↗</a>`;
        toast('Carga finalizada','YouTube devolvió el identificador del video.');
      }catch(error){
        $('uploadProgressText').textContent=`Error de carga: ${error.message}`;
        toast('No se completó la carga',error.message,'error');
      }finally{
        $('youtubeUpload').disabled=false;
      }
    }
